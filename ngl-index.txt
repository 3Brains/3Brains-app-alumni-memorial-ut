const { useState, useEffect } = React;

// ── Configuration ─────────────────────────────────────────────────────────────
const NGL_BASE = 'https://memorial-watch-backend-2.onrender.com';
const APP_VERSION = '1.0.0';
const APP_TITLE = 'VA Gravesite Tester';
const APP_SUBTITLE = 'Nationwide Gravesite Locator';

const isPhone  = window.innerWidth < 500;
const isTablet = window.innerWidth >= 500 && window.innerWidth < 1024;
const PT = isPhone ? '41vh' : isTablet ? '46vh' : '44vh';

// ── Helpers ───────────────────────────────────────────────────────────────────

const extractYear = (dateStr) => {
  if (!dateStr) return null;
  const m = String(dateStr).match(/\b(19|20)\d{2}\b/);
  return m ? m[0] : null;
};

const toTitleCase = (str) => {
  if (!str) return str;
  return str
    .replace(/\w\S*/g, (txt) => txt.charAt(0).toUpperCase() + txt.substr(1).toLowerCase())
    .replace(/'\w/g, (txt) => txt.toUpperCase());
};

// Calculate age at death from MM/DD/YYYY strings
const ageAtDeath = (dob, dod) => {
  if (!dob || !dod) return null;
  const by = extractYear(dob);
  const dy = extractYear(dod);
  if (!by || !dy) return null;
  var age = parseInt(dy) - parseInt(by);
  if (age < 0 || age > 120) return null;
  return age;
};

// Branch display cleanup (title case, but keep US intact)
const cleanBranch = (b) => {
  if (!b) return '';
  return b.split(',').map(function(part) {
    return part.trim().replace(/\b(US|USA|USAF|USMC|USN|USCG)\b/gi, function(m) {
      return m.toUpperCase();
    }).replace(/\b(AIR FORCE|ARMY|NAVY|MARINE CORPS|COAST GUARD|SPACE FORCE)\b/gi, function(m) {
      return toTitleCase(m);
    });
  }).join(', ');
};

const cleanWar = (w) => {
  if (!w) return '';
  return w.split(',').map(function(part) {
    return toTitleCase(part.trim().toLowerCase()).replace(/\bIi\b/g, 'II').replace(/\bI\b/g, 'I');
  }).join(', ');
};

const cleanCemeteryName = (c) => {
  if (!c) return '';
  return toTitleCase(c.toLowerCase());
};

const relationshipLabel = (r) => {
  if (!r) return 'Record';
  const lc = r.toLowerCase();
  if (lc.indexOf('self') > -1 || lc.indexOf('veteran') > -1) return 'Veteran';
  if (lc.indexOf('wife') > -1 || lc.indexOf('husband') > -1 || lc.indexOf('spouse') > -1) return 'Spouse';
  if (lc.indexOf('son') > -1 || lc.indexOf('daughter') > -1 || lc.indexOf('child') > -1) return 'Child';
  if (lc.indexOf('parent') > -1 || lc.indexOf('mother') > -1 || lc.indexOf('father') > -1) return 'Parent';
  return toTitleCase(r);
};

const relationshipColor = (r) => {
  const label = relationshipLabel(r);
  if (label === 'Veteran')  return 'bg-yellow-900/60 text-yellow-200 border-yellow-700/50';
  if (label === 'Spouse')   return 'bg-purple-900/60 text-purple-200 border-purple-700/50';
  if (label === 'Child')    return 'bg-blue-900/60 text-blue-200 border-blue-700/50';
  if (label === 'Parent')   return 'bg-green-900/60 text-green-200 border-green-700/50';
  return 'bg-gray-800 text-gray-300 border-gray-600';
};

// ── Icons ─────────────────────────────────────────────────────────────────────

const SearchIcon  = () => <svg className="w-7 h-7" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M21 21l-6-6m2-5a7 7 0 11-14 0 7 7 0 0114 0z"/></svg>;
const BookIcon    = () => <svg className="w-7 h-7" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M9 12h6m-6 4h6m2 5H7a2 2 0 01-2-2V5a2 2 0 012-2h5.586a1 1 0 01.707.293l5.414 5.414a1 1 0 01.293.707V19a2 2 0 01-2 2z"/></svg>;
const InfoIcon    = () => <svg className="w-7 h-7" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M13 16h-1v-4h-1m1-4h.01M21 12a9 9 0 11-18 0 9 9 0 0118 0z"/></svg>;
const PlusIcon    = () => <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M12 4v16m8-8H4"/></svg>;
const TrashIcon   = () => <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M19 7l-.867 12.142A2 2 0 0116.138 21H7.862a2 2 0 01-1.995-1.858L5 7m5 4v6m4-6v6m1-10V4a1 1 0 00-1-1h-4a1 1 0 00-1 1v3M4 7h16"/></svg>;
const CheckIcon   = () => <svg className="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M5 13l4 4L19 7"/></svg>;
const FlagIcon    = () => <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M3 21v-4m0 0V5a2 2 0 012-2h6.5l1 1H21l-3 6 3 6h-8.5l-1-1H5a2 2 0 00-2 2zm9-13.5V9"/></svg>;

// ── Common UI ─────────────────────────────────────────────────────────────────

const ClearableInput = ({ value, onChange, onBlur, onKeyPress, placeholder, type='text', maxLength, inputClass, numeric, autoCap }) => (
  <div className="relative">
    <input type={numeric ? 'tel' : type} placeholder={placeholder} value={value} onChange={onChange} onBlur={onBlur} onKeyPress={onKeyPress} maxLength={maxLength}
      className={inputClass + ' pr-10'}
      autoCorrect="off" autoCapitalize={autoCap || 'words'} autoComplete="off" spellCheck="false" />
    {value && value.length > 0 && (
      <button onClick={() => onChange({ target: { value: '' } })} className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-400 hover:text-white text-xl font-bold leading-none" type="button">X</button>
    )}
  </div>
);

const RelationshipBadge = ({ r }) => {
  const label = relationshipLabel(r);
  const color = relationshipColor(r);
  return (
    <span className={'inline-flex items-center gap-1 text-xs font-bold tracking-widest uppercase px-2 py-0.5 rounded-md border ' + color}>
      {label === 'Veteran' && <FlagIcon />}
      {label}
    </span>
  );
};

const BG_STYLE = {
  backgroundColor: '#0f1420',
  backgroundImage: 'linear-gradient(180deg, #1a1f2e 0%, #0f1420 60%, #0a0e18 100%)',
};

const DARK_STYLE = {
  backgroundColor: '#0a0e18',
  backgroundImage: 'linear-gradient(180deg, #111827 0%, #0a0e18 70%)',
};

const PageWrapper = ({ children, bgStyle }) => {
  var bg = bgStyle || BG_STYLE;
  return (
    <div style={{ minHeight:'100dvh', width:'100%', position:'relative', paddingBottom:'5rem',
      backgroundColor: bg.backgroundColor, backgroundImage: bg.backgroundImage || 'none' }}>
      <div style={{ position:'relative', zIndex:10 }}>{children}</div>
    </div>
  );
};

// ── NGL Result Card (reused in search results and memories drawer) ────────────

const NglResultCard = ({ r, onSave, alreadySaved }) => {
  const fullName = r.name || ((r.first_name || '') + ' ' + (r.last_name || '')).trim();
  const age = ageAtDeath(r.birth_date, r.death_date);
  const cem = cleanCemeteryName(r.cem_name);
  const branch = cleanBranch(r.branch);
  const war = cleanWar(r.war);
  const zip = r.zip || '';

  // Address line — skip components that are missing
  const addrLine1 = r.cem_addr_one || '';
  const addrLine2 = r.cem_addr_two || '';
  const cityStateZip = [r.city ? toTitleCase(r.city) : '', r.state].filter(Boolean).join(', ') + (zip ? ' ' + zip : '');

  // Buried-with info for non-self relationships
  const relLabel = relationshipLabel(r.relationship);
  const buriedWithName = (relLabel !== 'Veteran')
    ? [r.v_first_name, r.v_mid_name, r.v_last_name, r.v_suffix].filter(Boolean).join(' ').trim()
    : '';

  return (
    <div className="bg-black/50 border border-yellow-900/30 rounded-xl p-3 mb-2">
      <div className="flex items-start justify-between gap-2 mb-1">
        <RelationshipBadge r={r.relationship} />
        {typeof r._score === 'number' && r._score < 1 && (
          <span className="text-xs text-gray-500 italic">~match</span>
        )}
      </div>
      <h3 className="text-white text-xl font-bold leading-snug mb-1">{fullName}</h3>

      {buriedWithName && (
        <p className="text-purple-300 text-sm mb-1">Buried with veteran: {buriedWithName}</p>
      )}

      {(branch || r.rank || war) && (
        <p className="text-yellow-200 text-sm mb-1">
          {[branch, r.rank, war].filter(Boolean).join(' · ')}
        </p>
      )}

      <div className="text-gray-300 text-base leading-snug mb-2">
        {r.birth_date && <div>Born: {r.birth_date}</div>}
        {r.death_date && <div>Died: {r.death_date}{age ? ' (age ' + age + ')' : ''}</div>}
      </div>

      {cem && (
        <div className="bg-gray-900/60 rounded-lg p-2 mt-2">
          <p className="text-white font-semibold text-base">{cem}</p>
          {addrLine1 && <p className="text-gray-400 text-sm">{toTitleCase(addrLine1.toLowerCase())}</p>}
          {addrLine2 && <p className="text-gray-400 text-sm">{toTitleCase(addrLine2.toLowerCase())}</p>}
          {cityStateZip && <p className="text-gray-400 text-sm">{cityStateZip}</p>}
          {r.cem_phone && <p className="text-gray-400 text-sm">Phone: {r.cem_phone}</p>}
          {r.cem_url && (
            <p className="text-sm mt-1">
              <a href={r.cem_url} target="_blank" rel="noopener noreferrer" className="text-yellow-300 underline decoration-dotted">Cemetery details</a>
            </p>
          )}
        </div>
      )}

      {onSave && (
        alreadySaved
          ? <p className="text-green-400 text-sm font-bold text-center mt-3">✓ Saved to Memories</p>
          : <button onClick={() => onSave(r)}
              className="w-full py-3 mt-3 bg-gray-700 hover:bg-gray-600 border border-gray-500 text-white rounded-xl font-bold text-lg flex items-center justify-center gap-2">
              <PlusIcon /> Save to Memories
            </button>
      )}
    </div>
  );
};

// ── Main App ──────────────────────────────────────────────────────────────────

function App() {
  const [page,         setPage]         = useState('search');
  const [apiVersion,   setApiVersion]   = useState('...');
  const [memories,     setMemories]     = useState(() => {
    try { return JSON.parse(localStorage.getItem('ngl_memories') || '[]'); } catch(e) { return []; }
  });
  const [nw,           setNw]           = useState({ firstName:'', lastName:'', middle:'', birthYear:'' });
  const [searchCheck,  setSearchCheck]  = useState(null);
  const [searching,    setSearching]    = useState(false);
  const [searched,     setSearched]     = useState(false);
  const [debugOpen,    setDebugOpen]    = useState(false);
  const [loadingMore,  setLoadingMore]  = useState(false);

  useEffect(() => {
    fetch(NGL_BASE + '/health').then(r => r.json()).then(d => setApiVersion(d.version || '?')).catch(() => setApiVersion('?'));
  }, []);

  const saveMemories = (list) => {
    setMemories(list);
    localStorage.setItem('ngl_memories', JSON.stringify(list));
  };

  const isInMemories = (decedent_id) => {
    if (!decedent_id) return false;
    return memories.some(m => m.decedent_id === decedent_id);
  };

  const addToMemories = (r) => {
    if (isInMemories(r.decedent_id)) return;
    const entry = Object.assign({}, r, { added: new Date().toISOString() });
    saveMemories([entry].concat(memories));
  };

  const delMemory = (idx) => saveMemories(memories.filter((m, i) => i !== idx));

  const clearSearch = () => {
    setSearchCheck(null); setSearched(false); setDebugOpen(false);
    setNw({ firstName:'', lastName:'', middle:'', birthYear:'' });
  };

  const runSearch = async (offsetOverride) => {
    if (document.activeElement) document.activeElement.blur();
    const firstName = toTitleCase((nw.firstName || '').trim());
    const lastName  = toTitleCase((nw.lastName  || '').trim());
    if (!lastName || lastName.length < 2) { alert('Last name is required.'); return; }

    const offset = offsetOverride || 0;
    if (offset === 0) { setSearching(true); setSearched(false); setSearchCheck(null); setDebugOpen(false); }
    else { setLoadingMore(true); }

    const params = new URLSearchParams();
    params.set('last_name', lastName);
    if (firstName)     params.set('first_name',  firstName);
    if (nw.middle)     params.set('middle_name', nw.middle.trim());
    if (nw.birthYear)  params.set('birth_year',  nw.birthYear.trim());
    params.set('offset', String(offset));

    const url = NGL_BASE + '/ngl/proxy?' + params.toString();
    var data = { results: [], has_more: false };
    try {
      const resp = await fetch(url);
      if (resp.ok) data = await resp.json();
    } catch(e) { console.log('ngl error:', e); }

    if (offset === 0) {
      setSearchCheck({
        firstName, lastName,
        results: data.results || [],
        hasMore: !!data.has_more,
        debugData: { url: url, count: (data.results || []).length, has_more: !!data.has_more, error: data.error || null }
      });
    } else if (searchCheck) {
      setSearchCheck(Object.assign({}, searchCheck, {
        results: searchCheck.results.concat(data.results || []),
        hasMore: !!data.has_more
      }));
    }
    setSearched(true);
    setSearching(false);
    setLoadingMore(false);
  };

  const loadMore = () => {
    if (!searchCheck) return;
    runSearch(searchCheck.results.length);
  };

  const iClass = "w-full px-4 py-3 border border-yellow-900/50 rounded-lg focus:ring-2 focus:ring-yellow-500 text-lg bg-gray-900/60 text-white placeholder-gray-400";
  const cardClass = "bg-black/35 rounded-2xl p-4 shadow-lg border border-yellow-900/30 backdrop-blur-sm";

  // ── Pages ───────────────────────────────────────────────────────────────────

  const SearchPage = () => (
    <div className="space-y-3" style={{ paddingTop:'1rem' }}>
      <div className={cardClass}>
        <div className="mb-1 pb-1 border-b border-yellow-900/30 text-center">
          <p className="text-yellow-300 text-3xl font-bold italic whitespace-nowrap">Search VA Records</p>
          <p className="text-gray-400 text-sm mt-1">Veterans and their families</p>
        </div>
        {searching && <p className="text-white text-center py-4 animate-pulse">Searching...</p>}
        {!searched ? (
          <div className="space-y-3">
            <ClearableInput value={nw.firstName}
              onChange={(e) => setNw(Object.assign({}, nw, { firstName: e.target.value }))}
              onBlur={(e) => setNw(Object.assign({}, nw, { firstName: toTitleCase(e.target.value.trim()) }))}
              onKeyPress={(e) => e.key==='Enter' && runSearch(0)}
              placeholder="First Name"
              inputClass={iClass} />
            <ClearableInput value={nw.lastName}
              onChange={(e) => setNw(Object.assign({}, nw, { lastName: e.target.value }))}
              onBlur={(e) => setNw(Object.assign({}, nw, { lastName: toTitleCase(e.target.value.trim()) }))}
              onKeyPress={(e) => e.key==='Enter' && runSearch(0)}
              placeholder="Last Name *"
              inputClass={iClass} />
            <ClearableInput value={nw.middle}
              onChange={(e) => setNw(Object.assign({}, nw, { middle: e.target.value }))}
              onBlur={(e) => { var v = e.target.value.trim(); if (v.length === 1) setNw(Object.assign({}, nw, { middle: v.toUpperCase() })); }}
              onKeyPress={(e) => e.key==='Enter' && runSearch(0)}
              placeholder="Middle Name or Initial"
              inputClass={iClass} />
            <ClearableInput value={nw.birthYear}
              onChange={(e) => setNw(Object.assign({}, nw, { birthYear: e.target.value.replace(/\D/g,'').slice(0,4) }))}
              onKeyPress={(e) => e.key==='Enter' && runSearch(0)}
              placeholder="Birth Year" maxLength={4}
              inputClass={iClass} numeric={true} />
            <button onClick={() => runSearch(0)} disabled={searching}
              className="w-full py-3 bg-yellow-600 text-black rounded-lg flex items-center justify-center gap-2 disabled:opacity-50 text-xl font-bold">
              <SearchIcon /> Search
            </button>
            <p className="text-gray-400 text-sm text-center pt-1">
              Catches typos automatically (e.g. "Norman" finds "Normal")
            </p>
          </div>
        ) : (
          <SearchResults />
        )}
      </div>
    </div>
  );

  const SearchResults = () => {
    if (!searchCheck) return null;
    const { results, hasMore, debugData, firstName, lastName } = searchCheck;

    return (
      <div className="space-y-3">
        <div className="flex items-center justify-between pt-1">
          <p className="text-yellow-300 text-lg font-bold">
            {results.length} result{results.length === 1 ? '' : 's'}
          </p>
          <button onClick={clearSearch} className="text-white/60 underline decoration-dotted text-sm">
            New search
          </button>
        </div>

        {results.length === 0 && (
          <div className="rounded-2xl overflow-hidden border border-yellow-800/40">
            <div className="px-4 py-3 bg-gray-800 text-center">
              <p className="text-white font-black text-xl">{(firstName + ' ' + lastName).trim()}</p>
              <p className="text-white/80 text-lg">Not Found in VA Records</p>
            </div>
            <div className="bg-gray-900/70 p-4">
              <p className="text-gray-300 text-base leading-relaxed text-center">
                No matching record in the VA Nationwide Gravesite Locator.
                The person may not be a veteran, may not be buried in a VA cemetery,
                or may have a different name spelling.
              </p>
            </div>
          </div>
        )}

        {results.length > 0 && results.map((r, i) => (
          <NglResultCard key={r.decedent_id || i} r={r}
            onSave={addToMemories}
            alreadySaved={isInMemories(r.decedent_id)} />
        ))}

        {hasMore && (
          <button onClick={loadMore} disabled={loadingMore}
            className="w-full py-2.5 bg-gray-700 border border-yellow-800/50 text-white rounded-xl text-lg font-bold disabled:opacity-50">
            {loadingMore ? 'Loading...' : 'Show 10 more'}
          </button>
        )}

        <button onClick={clearSearch}
          className="w-full py-2 bg-white/10 border border-white/20 text-white/80 rounded-xl text-lg mt-2">
          Search for another name
        </button>

        {debugData && (
          <div className="mt-3">
            <button onClick={() => setDebugOpen(!debugOpen)}
              className="w-full flex justify-between items-center px-3 py-2 bg-black/40 border border-white/10 text-white/30 rounded-t-lg text-xs uppercase tracking-widest">
              <span>Raw API Response</span><span>{debugOpen ? 'HIDE' : 'SHOW'}</span>
            </button>
            {debugOpen && (
              <pre className="bg-black/90 border border-white/10 border-t-0 rounded-b-lg p-3 text-xs text-green-400 overflow-auto max-h-56 whitespace-pre-wrap break-all">
                {JSON.stringify(debugData, null, 2)}
              </pre>
            )}
          </div>
        )}
      </div>
    );
  };

  const MemoriesPage = () => (
    <div className="space-y-3" style={{ paddingTop:'1rem' }}>
      <div className={cardClass}>
        <div className="mb-3 pb-2 border-b border-yellow-900/30">
          <div className="flex items-center justify-between">
            <div className="w-10" />
            <h2 className="text-3xl font-black text-white tracking-wide text-center flex-1">Memories</h2>
            <button onClick={() => { clearSearch(); setPage('search'); }}
              className="flex flex-col items-center text-white hover:text-yellow-400 transition-colors w-10">
              <SearchIcon /><span className="text-xs">Search</span>
            </button>
          </div>
          <p className="text-gray-300 text-base mt-1 text-center">Saved VA records</p>
        </div>

        {memories.length === 0 ? (
          <div className="text-center py-6">
            <p className="text-gray-300 text-lg leading-relaxed">No saved records yet.</p>
            <p className="text-yellow-300 text-base font-medium mt-2">Use Search to find and save people.</p>
          </div>
        ) : (
          <div className="space-y-2">
            {memories.map((m, i) => (
              <div key={m.decedent_id || i} className="relative">
                <NglResultCard r={m} />
                <button onClick={() => delMemory(i)}
                  className="absolute top-3 right-3 p-2 text-red-500 hover:bg-red-900/30 rounded-lg">
                  <TrashIcon />
                </button>
              </div>
            ))}
          </div>
        )}
      </div>
    </div>
  );

  const InfoPage = () => (
    <div className="space-y-4 pb-8" style={{ paddingTop:'1rem' }}>
      <div className={cardClass}>
        <div className="mb-3 pb-2 border-b border-yellow-900/30 text-center">
          <h2 className="text-3xl font-black text-white tracking-wide">{APP_TITLE}</h2>
          <p className="text-yellow-300 italic text-lg mt-1">v{APP_VERSION}</p>
        </div>
        <div className="space-y-3 text-gray-200 text-lg leading-relaxed">
          <p>Standalone tester for the VA Nationwide Gravesite Locator (NGL).
             Calls the shared Memory Watch backend, which wraps the Socrata
             open-data API for NGL.</p>
          <p>Every record represents a deceased veteran or a family member
             buried in a VA cemetery. The relationship badge distinguishes
             veterans (self) from spouses, children, and other family members.</p>
          <p className="text-yellow-300 text-base">
             Typo tolerance is built into the backend — searching for
             "Norman" will still find a record incorrectly entered as "Normal".
          </p>
        </div>
        <div className="mt-4 p-3 bg-black/30 rounded-xl space-y-1">
          <p className="text-gray-400 text-sm text-center">app v{APP_VERSION} | api {apiVersion}</p>
          <p className="text-gray-500 text-xs text-center">Backend: {NGL_BASE}</p>
          <p className="text-gray-500 text-xs text-center">Data: data.va.gov/resource/3u66-fxug</p>
        </div>
        <button onClick={() => {
          if (window.confirm('Clear all saved Memories? This cannot be undone.')) {
            localStorage.removeItem('ngl_memories');
            saveMemories([]);
            alert('All data cleared.');
          }
        }} className="w-full py-3 mt-3 bg-red-900/40 border border-red-800/50 text-red-300 rounded-xl text-lg font-bold">
          Clear All Test Data
        </button>
      </div>
    </div>
  );

  // ── Render ──────────────────────────────────────────────────────────────────

  return (
    <PageWrapper bgStyle={page === 'memories' ? DARK_STYLE : BG_STYLE}>
      <div className="text-center pt-4 pb-1">
        <p className="text-yellow-400 text-xl font-bold tracking-widest">{APP_TITLE}</p>
        <p className="text-xs tracking-widest uppercase text-white/50 mt-0.5">{APP_SUBTITLE}</p>
      </div>

      <div className="w-full px-3">
        {page === 'search'   && <SearchPage />}
        {page === 'memories' && <MemoriesPage />}
        {page === 'info'     && <InfoPage />}
      </div>

      <div style={{ position:'fixed', bottom:0, left:0, right:0,
        backgroundColor:'rgba(10,10,20,0.95)', backdropFilter:'blur(8px)',
        borderTop:'2px solid #c8a951', padding:'0.5rem 1rem',
        paddingBottom:'max(0.5rem, env(safe-area-inset-bottom))' }}>
        <div className="flex justify-around max-w-sm mx-auto">
          <button onClick={() => { clearSearch(); setPage('search'); }}
            className={'flex flex-col items-center gap-0.5 px-1 py-1 ' + (page==='search' ? 'text-yellow-500' : 'text-gray-400')}>
            <SearchIcon /><span className="text-lg font-medium">Search</span>
          </button>
          <button onClick={() => setPage('memories')}
            className={'flex flex-col items-center gap-0.5 px-1 py-1 ' + (page==='memories' ? 'text-yellow-500' : 'text-gray-400')}>
            <BookIcon /><span className="text-lg font-medium">Memories</span>
          </button>
          <button onClick={() => setPage('info')}
            className={'flex flex-col items-center gap-0.5 px-1 py-1 ' + (page==='info' ? 'text-yellow-500' : 'text-gray-400')}>
            <InfoIcon /><span className="text-lg font-medium">Info</span>
          </button>
        </div>
        <p className="text-center text-gray-600 text-xs mt-1" style={{ display: page==='search' ? 'none' : 'block' }}>
          app v{APP_VERSION} | api {apiVersion}
        </p>
      </div>
    </PageWrapper>
  );
}
